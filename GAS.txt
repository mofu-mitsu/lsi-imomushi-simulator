 * 🐛 LSI芋虫 育成・統制シミュレーター用 Google Apps Script (GAS)
 * 
 * 【Googleアカウント別 & ゲスト個別データ完全保存・アイテム復元対応版 (v3)】
 * 
 * 【導入手順】
 * 1. お手持ちの Google スプレッドシート を開きます。
 * 2. 上部メニューの「拡張機能」 > 「Apps Script」を開きます。
 * 3. エディタ内のコードをすべて削除し、このファイルの内容をすべて貼り付けて保存（Ctrl+S / Cmd+S）します。
 * 4. 画面右上の「デプロイ」 > 「新しいデプロイ」をクリックします。
 * 5. 種類を「ウェブアプリ」にし、以下を設定します：
 *    - 次のユーザーとして実行: 「自分」
 *    - アクセスできるユーザー: 「全員」
 * 6. 「デプロイ」を押して表示された「ウェブアプリ URL」をコピーし、
 *    LSI芋虫アプリに入力するか、DEFAULT_GAS_URLに設定すれば連携完了です！
 * 
 * ※ 既に古いシートが存在する場合でも、自動で新しい列定義に合わせてアップグレードされます。
 */

function doGet(e) {
  try {
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    initSheetsIfNeeded(ss);
    
    const action = (e && e.parameter && e.parameter.action) || 'getStatus';
    const uid = (e && e.parameter && e.parameter.uid) || '';
    
    if (action === 'getStatus' || action === 'getUserStatus') {
      const sheet = ss.getSheetByName('個体ステータス_アカウント別');
      if (!sheet) {
        return createJsonResponse({ success: false, error: 'Sheet not found' });
      }
      
      const data = sheet.getDataRange().getValues();
      
      // Look for row matching requested UID
      if (uid && data.length > 1) {
        for (let i = 1; i < data.length; i++) {
          if (String(data[i][0]) === String(uid)) {
            const userObj = parseRowToUserObject(data[i]);
            return createJsonResponse({ success: true, data: userObj });
          }
        }
      }
      
      // If UID not found, return null so the frontend knows it's a new user
      return createJsonResponse({ success: true, data: null });
    }
    
    if (action === 'getAnnouncements') {
      const sheet = ss.getSheetByName('お知らせ');
      if (!sheet) {
        return createJsonResponse({ success: true, data: [] });
      }
      const data = sheet.getDataRange().getValues();
      const announcements = [];
      // skip header row 0
      for (let i = 1; i < data.length; i++) {
        if (data[i][0]) {
          announcements.push({
            id: String(data[i][0]),
            date: String(data[i][1]),
            title: String(data[i][2]),
            content: String(data[i][3])
          });
        }
      }
      return createJsonResponse({ success: true, data: announcements.reverse() }); // newest first
    }
    
    return createJsonResponse({ success: true, message: 'LSI Caterpillar GAS Backend (v4) is running.' });
  } catch (error) {
    return createJsonResponse({ success: false, error: error.toString() });
  }
}

function doPost(e) {
  try {
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    initSheetsIfNeeded(ss);
    
    const postData = JSON.parse(e.postData.contents);
    const action = postData.action;
    const nowStr = Utilities.formatDate(new Date(), 'Asia/Tokyo', 'yyyy/MM/dd HH:mm:ss');
    
    if (action === 'saveStatus') {
      const sheet = ss.getSheetByName('個体ステータス_アカウント別');
      const data = postData.data;
      const targetUid = String(data.uid || 'guest_default');
      
      const values = sheet.getDataRange().getValues();
      let targetRowIndex = -1;
      
      // Check existing UID row
      for (let i = 1; i < values.length; i++) {
        if (String(values[i][0]) === targetUid) {
          targetRowIndex = i + 1;
          break;
        }
      }
      
      const rowData = [
        targetUid,                                            // 1: UID
        data.ownerName || '未設定',                           // 2: 飼育員名
        data.selfType || '未設定',                            // 3: 自認タイプ
        data.id || '000001',                                 // 4: 個体ID
        data.name || 'LSI芋虫',                              // 5: 名前・肩書
        data.stage !== undefined ? data.stage : 0,           // 6: 進化段階
        data.exp !== undefined ? data.exp : 0,               // 7: 獲得EXP
        data.points !== undefined ? data.points : 0,         // 8: 統制ポイント
        JSON.stringify(data.furniture || []),                // 9: 所有家具
        JSON.stringify(data.discoveredStages || []),         // 10: アンロック図鑑
        data.sprayCount !== undefined ? data.sprayCount : 0, // 11: 防虫スプレー所持数
        data.daycareUntil || '',                             // 12: 保育園保護期限
        JSON.stringify(data.foodStats || { cabbage: 0, apple: 0, glucose: 0, sugar: 0, twig: 0 }), // 13: エサ摂取統計
        data.formVariant || 'crystal',                       // 14: 形態バリアント
        data.darlingIncident ? true : false,                 // 15: ダーリンインシデント
        data.squashCount !== undefined ? data.squashCount : 0, // 16: 潰された回数
        data.lastFedAt || nowStr,                            // 17: 最終給餌日時
        data.lastMessageAt || nowStr,                        // 18: 最終交信日時
        nowStr                                               // 19: 最終更新日時
      ];
      
      if (targetRowIndex > 0) {
        // Update existing row
        sheet.getRange(targetRowIndex, 1, 1, rowData.length).setValues([rowData]);
      } else {
        // Append new row for this user / guest
        sheet.appendRow(rowData);
      }
      
      return createJsonResponse({ success: true, message: 'Status saved for UID: ' + targetUid });
    }
    
    if (action === 'appendLog') {
      const sheet = ss.getSheetByName('行動ログ');
      sheet.appendRow([
        nowStr,
        postData.uid || 'guest',
        postData.type || '一般',
        postData.detail || '',
        postData.expChange || 0
      ]);
      return createJsonResponse({ success: true, message: 'Log appended.' });
    }
    
    if (action === 'appendChat') {
      const sheet = ss.getSheetByName('交信記録');
      sheet.appendRow([
        nowStr,
        postData.uid || 'guest',
        postData.sender || '発言者',
        postData.message || ''
      ]);
      return createJsonResponse({ success: true, message: 'Chat recorded.' });
    }
    
    return createJsonResponse({ success: false, error: 'Unknown action: ' + action });
  } catch (error) {
    return createJsonResponse({ success: false, error: error.toString() });
  }
}

function parseRowToUserObject(row) {
  return {
    uid: row[0],
    ownerName: row[1],
    selfType: row[2],
    id: row[3],
    name: row[4],
    stage: Number(row[5]) || 0,
    exp: Number(row[6]) || 0,
    points: Number(row[7]) || 0,
    furniture: safeJsonParse(row[8], []),
    discoveredStages: safeJsonParse(row[9], ['LSI芋虫（幼虫）']),
    sprayCount: Number(row[10]) || 0,
    daycareUntil: row[11] || null,
    foodStats: safeJsonParse(row[12], { cabbage: 0, apple: 0, glucose: 0, sugar: 0, twig: 0 }),
    formVariant: row[13] || 'crystal',
    darlingIncident: row[14] === true || String(row[14]) === 'true',
    squashCount: Number(row[15]) || 0,
    lastFedAt: row[16] || '',
    lastMessageAt: row[17] || ''
  };
}

function safeJsonParse(str, fallback) {
  if (!str) return fallback;
  try {
    return JSON.parse(str);
  } catch (e) {
    return fallback;
  }
}

function initSheetsIfNeeded(ss) {
  const headers = [
    'UID (アカウント識別子)', 
    '飼育員名', 
    '自認タイプ', 
    '個体ID', 
    '名前・肩書', 
    '進化段階', 
    '獲得EXP', 
    '統制ポイント', 
    '所有家具 (JSON)', 
    '設置中家具 (JSON)',
    'アンロック図鑑 (JSON)',
    '防虫スプレー回数',
    '保育園保護期限',
    'エサ摂取統計 (JSON)',
    '形態バリアント',
    'ダーリンインシデント',
    '潰された回数',
    '最終給餌日時', 
    '最終交信日時', 
    '最終更新日時'
  ];

  let statusSheet = ss.getSheetByName('個体ステータス_アカウント別');
  if (!statusSheet) {
    statusSheet = ss.insertSheet('個体ステータス_アカウント別');
    statusSheet.getRange(1, 1, 1, headers.length).setValues([headers]);
    statusSheet.getRange(1, 1, 1, headers.length).setBackground('#d1fae5').setFontWeight('bold');
  } else {
    const currentCols = statusSheet.getLastColumn();
    // Upgrade existing data to 20 columns if they only have 19 or less
    if (currentCols < headers.length) {
      // First, get all data
      const data = statusSheet.getDataRange().getValues();
      
      // Update header
      statusSheet.getRange(1, 1, 1, headers.length).setValues([headers]);
      statusSheet.getRange(1, 1, 1, headers.length).setBackground('#d1fae5').setFontWeight('bold');
      
      // We must insert a blank column for "設置中家具" at column 10 (index 9) for all existing rows 
      // if they only had 19 columns. But wait, if they have less columns, their data index is shifted!
      // To be safe, we will just rely on parseRowToUserObject checking row length on read, and saveStatus will overwrite with 20 columns.
      // But it's better to just insert the missing column so formatting aligns.
      if (currentCols === 19) {
        statusSheet.insertColumnBefore(10);
        statusSheet.getRange(1, 10).setValue('設置中家具 (JSON)');
      }
    }
  }
  
  let noticeSheet = ss.getSheetByName('お知らせ');
  if (!noticeSheet) {
    noticeSheet = ss.insertSheet('お知らせ');
    const noticeHeaders = ['ID', '日付', 'タイトル', '本文(改行可)'];
    noticeSheet.getRange(1, 1, 1, noticeHeaders.length).setValues([noticeHeaders]);
    noticeSheet.getRange(1, 1, 1, noticeHeaders.length).setBackground('#fde68a').setFontWeight('bold');
    
    // Add sample announcement
    noticeSheet.appendRow(['msg_001', Utilities.formatDate(new Date(), 'Asia/Tokyo', 'yyyy/MM/dd'), 'LSI規律お知らせ機能追加', 'お知らせをスプレッドシートから配信できるようになりました。']);
  }
  let logSheet = ss.getSheetByName('行動ログ');
  if (!logSheet) {
    logSheet = ss.insertSheet('行動ログ');
    logSheet.getRange(1, 1, 1, 5).setValues([['タイムスタンプ', 'UID', '種別', '観測イベント内容', '獲得EXP']]);
    logSheet.getRange(1, 1, 1, 5).setBackground('#d1fae5').setFontWeight('bold');
  }
  
  let chatSheet = ss.getSheetByName('交信記録');
  if (!chatSheet) {
    chatSheet = ss.insertSheet('交信記録');
    chatSheet.getRange(1, 1, 1, 4).setValues([['日時', 'UID', '発信者', '発言内容']]);
    chatSheet.getRange(1, 1, 1, 4).setBackground('#d1fae5').setFontWeight('bold');
  }
}

function createJsonResponse(data) {
  return ContentService.createTextOutput(JSON.stringify(data))
    .setMimeType(ContentService.MimeType.JSON);
}
